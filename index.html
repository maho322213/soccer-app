const tagContainer = document.getElementById('tagContainer'); const newTagInput = document.getElementById('newTagInput'); const addTagBtn = document.getElementById('addTagBtn'); const highlightList = document.getElementById('highlightList'); let recordedTags = []; // 基本タグのクリック document.querySelectorAll('.tagBtn').forEach(btn => { btn.addEventListener('click', () => { const tagName = btn.getAttribute('data-tag'); const time = video.currentTime; recordedTags.push({ tag: tagName, time }); updateHighlights(); }); }); // オリジナルタグの追加 addTagBtn.addEventListener('click', () => { const tagName = newTagInput.value.trim(); if (!tagName) return; const newBtn = document.createElement('button'); newBtn.classList.add('tagBtn'); newBtn.setAttribute('data-tag', tagName); newBtn.textContent = tagName; tagContainer.appendChild(newBtn); newBtn.addEventListener('click', () => { const time = video.currentTime; recordedTags.push({ tag: tagName, time }); updateHighlights(); }); newTagInput.value = ''; }); // ハイライトの更新 function updateHighlights() { highlightList.innerHTML = ''; recordedTags.forEach((entry, index) => { const btn = document.createElement('button'); btn.textContent = `${entry.tag} (${formatTime(entry.time)})`; btn.addEventListener('click', () => playHighlight(entry.time)); highlightList.appendChild(btn); }); } // ハイライト再生（前後5秒） function playHighlight(time) { const start = Math.max(0, time - 5); const end = Math.min(video.duration, time + 5); video.currentTime = start; video.play(); const interval = setInterval(() => { if (video.currentTime >= end) { video.pause(); clearInterval(interval); } }, 100); } // 秒 → mm:ss 形式 function formatTime(seconds) { const m = Math.floor(seconds / 60); const s = Math.floor(seconds % 60); return `${m}:${s.toString().padStart(2, '0')}`; }
